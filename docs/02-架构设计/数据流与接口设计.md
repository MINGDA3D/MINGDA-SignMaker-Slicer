```python
from abc import ABC, abstractmethod
from typing import List

class IDXFParser(ABC):
    """DXF解析器接口"""

    @abstractmethod
    def parse(self, file_path: str) -> List[Contour]:
        """
        解析DXF文件

        Args:
            file_path: DXF文件路径

        Returns:
            解析得到的轮廓列表

        Raises:
            FileNotFoundError: 文件不存在
            DXFParseError: 解析失败
        """
        pass

    @abstractmethod
    def get_supported_entities(self) -> List[str]:
        """获取支持的实体类型"""
        pass

    @abstractmethod
    def validate(self, file_path: str) -> bool:
        """验证文件是否可解析"""
        pass
```

### 5.2 事件分发器

```python
from typing import Callable, Dict, List
from collections import defaultdict

class EventDispatcher:
    """事件分发器（单例）"""

    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._handlers = defaultdict(list)
        return cls._instance

    def subscribe(self, event_type: EventType,
                   handler: Callable[[Event], None]) -> None:
        """
        订阅事件

        Args:
            event_type: 事件类型
            handler: 处理函数
        """
        self._handlers[event_type].append(handler)

    def unsubscribe(self, event_type: EventType,
                     handler: Callable[[Event], None]) -> None:
        """取消订阅"""
        if handler in self._handlers[event_type]:
            self._handlers[event_type].remove(handler)

    def dispatch(self, event: Event) -> None:
        """
        分发事件

        Args:
            event: 要分发的事件
        """
        for handler in self._handlers[event.type]:
            try:
                handler(event)
            except Exception as e:
                # 记录错误但不中断其他处理器
                print(f"Event handler error: {e}")

    def clear(self, event_type: Optional[EventType] = None) -> None:
        """清除事件处理器"""
        if event_type:
            self._handlers[event_type].clear()
        else:
            self._handlers.clear()


# 全局事件分发器实例
event_dispatcher = EventDispatcher()
```

### 5.3 事件使用示例

```python
# 订阅事件
def on_slice_completed(event: Event):
    result = event.data
    print(f"切片完成，共 {result.total_layers} 层")

event_dispatcher.subscribe(EventType.SLICE_COMPLETED, on_slice_completed)

# 发布事件
def complete_slice(result: SliceResult):
    event = Event(
        type=EventType.SLICE_COMPLETED,
        source="SliceManager",
        data=result
    )
    event_dispatcher.dispatch(event)
```

## 6. 模块通信协议

### 6.1 请求-响应模式

```python
@dataclass
class Request:
    """请求基类"""
    id: str
    action: str
    params: Dict[str, Any] = field(default_factory=dict)

@dataclass
class Response:
    """响应基类"""
    request_id: str
    success: bool
    data: Optional[Any] = None
    error: Optional[str] = None


class RequestHandler:
    """请求处理器"""

    def __init__(self):
        self._handlers: Dict[str, Callable] = {}

    def register(self, action: str,
                  handler: Callable[[Request], Response]) -> None:
        """注册处理器"""
        self._handlers[action] = handler

    def handle(self, request: Request) -> Response:
        """处理请求"""
        if request.action not in self._handlers:
            return Response(
                request_id=request.id,
                success=False,
                error=f"Unknown action: {request.action}"
            )

        try:
            return self._handlers[request.action](request)
        except Exception as e:
            return Response(
                request_id=request.id,
                success=False,
                error=str(e)
            )
```

```python
from abc import ABC, abstractmethod

class Command(ABC):
    """命令基类（支持撤销/重做）"""

    @abstractmethod
    def execute(self) -> None:
        """执行命令"""
        pass

    @abstractmethod
    def undo(self) -> None:
        """撤销命令"""
        pass

    @property
    @abstractmethod
    def description(self) -> str:
        """命令描述"""
        pass


class CommandManager:
    """命令管理器"""

    def __init__(self, max_history: int = 100):
        self._history: List[Command] = []
        self._redo_stack: List[Command] = []
        self._max_history = max_history

    def execute(self, command: Command) -> None:
        """执行命令并记录历史"""
        command.execute()
        self._history.append(command)
        self._redo_stack.clear()

        # 限制历史记录数量
        if len(self._history) > self._max_history:
            self._history.pop(0)

    def undo(self) -> bool:
        """撤销上一个命令"""
        if not self._history:
            return False

        command = self._history.pop()
        command.undo()
        self._redo_stack.append(command)
        return True

    def redo(self) -> bool:
        """重做上一个撤销的命令"""
        if not self._redo_stack:
            return False

        command = self._redo_stack.pop()
        command.execute()
        self._history.append(command)
        return True

    @property
    def can_undo(self) -> bool:
        return len(self._history) > 0

    @property
    def can_redo(self) -> bool:
        return len(self._redo_stack) > 0
```

```python
from PySide6.QtCore import QThread, Signal, QMutex

class SliceWorker(QThread):
    """切片工作线程"""

    # 信号
    progress = Signal(int, str)
    completed = Signal(SliceResult)
    failed = Signal(str)
    cancelled = Signal()

    def __init__(self, engine: ISliceEngine,
                  contours: List[Contour],
                  params: SliceParams):
        super().__init__()
        self.engine = engine
        self.contours = contours
        self.params = params
        self._cancelled = False
        self._mutex = QMutex()

    def run(self):
        """执行切片"""
        try:
            # 设置进度回调
            def progress_callback(percent: int, message: str):
                if self._is_cancelled():
                    raise CancelledException()
                self.progress.emit(percent, message)

            self.engine.set_progress_callback(progress_callback)

            # 执行切片
            result = self.engine.slice(self.contours, self.params)

            if self._is_cancelled():
                self.cancelled.emit()
            else:
                self.completed.emit(result)

        except CancelledException:
            self.cancelled.emit()
        except Exception as e:
            self.failed.emit(str(e))

    def cancel(self):
        """取消切片"""
        self._mutex.lock()
        self._cancelled = True
        self._mutex.unlock()
        self.engine.cancel()

    def _is_cancelled(self) -> bool:
        self._mutex.lock()
        result = self._cancelled
        self._mutex.unlock()
        return result
```

### 7.2 任务队列

```python
from queue import Queue
from threading import Thread
from dataclasses import dataclass
from typing import Callable, Optional

@dataclass
class Task:
    """任务定义"""
    id: str
    func: Callable
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    callback: Optional[Callable] = None
    error_callback: Optional[Callable] = None


class TaskQueue:
    """任务队列"""

    def __init__(self, num_workers: int = 2):
        self._queue = Queue()
        self._workers: List[Thread] = []
        self._running = True

        for _ in range(num_workers):
            worker = Thread(target=self._worker_loop, daemon=True)
            worker.start()
            self._workers.append(worker)

    def submit(self, task: Task) -> None:
        """提交任务"""
        self._queue.put(task)

    def _worker_loop(self):
        """工作线程循环"""
        while self._running:
            try:
                task = self._queue.get(timeout=1.0)
            except:
                continue

            try:
                result = task.func(*task.args, **task.kwargs)
                if task.callback:
                    task.callback(result)
            except Exception as e:
                if task.error_callback:
                    task.error_callback(e)
            finally:
                self._queue.task_done()

    def shutdown(self):
        """关闭队列"""
        self._running = False
        for worker in self._workers:
            worker.join()
```

```python
from typing import Callable, Type
from functools import wraps

class ErrorHandler:
    """错误处理器"""

    def __init__(self):
        self._handlers: Dict[Type[Exception], Callable] = {}

    def register(self, exception_type: Type[Exception],
                  handler: Callable[[Exception], None]) -> None:
        """注册异常处理器"""
        self._handlers[exception_type] = handler

    def handle(self, exception: Exception) -> bool:
        """
        处理异常

        Returns:
            是否成功处理
        """
        for exc_type, handler in self._handlers.items():
            if isinstance(exception, exc_type):
                handler(exception)
                return True
        return False


def error_handler(fallback_value=None):
    """错误处理装饰器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except SlicerError as e:
                # 记录错误
                print(f"Slicer error in {func.__name__}: {e}")
                # 发送错误事件
                event_dispatcher.dispatch(Event(
                    type=EventType.ERROR_OCCURRED,
                    source=func.__name__,
                    data={"error": e, "args": args, "kwargs": kwargs}
                ))
                return fallback_value
            except Exception as e:
                # 未预期的错误
                print(f"Unexpected error in {func.__name__}: {e}")
                raise
        return wrapper
    return decorator
```

```python
import json
from pathlib import Path

@dataclass
class AppConfig:
    """应用配置"""

    # 界面
    language: str = "zh_CN"
    theme: str = "dark"
    window_geometry: Optional[bytes] = None
    window_state: Optional[bytes] = None

    # 路径
    last_open_dir: str = ""
    last_save_dir: str = ""
    last_export_dir: str = ""

    # 默认参数
    default_machine: str = "ZC-60"
    default_layer_height: float = 0.3
    default_wall_thickness: float = 1.2

    # 性能
    max_undo_history: int = 100
    auto_save_interval: int = 300  # 秒
    enable_hardware_acceleration: bool = True

    # 最近文件
    recent_files: List[str] = field(default_factory=list)
    max_recent_files: int = 10


class ConfigManager:
    """配置管理器"""

    def __init__(self, config_path: str = "config.json"):
        self.config_path = Path(config_path)
        self.config = AppConfig()
        self.load()

    def load(self) -> None:
        """加载配置"""
        if self.config_path.exists():
            try:
                with open(self.config_path, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    for key, value in data.items():
                        if hasattr(self.config, key):
                            setattr(self.config, key, value)
            except Exception as e:
                print(f"Failed to load config: {e}")

    def save(self) -> None:
        """保存配置"""
        try:
            data = {
                key: getattr(self.config, key)
                for key in self.config.__dataclass_fields__
                if not key.startswith('_')
            }
            with open(self.config_path, 'w', encoding='utf-8') as f:
                json.dump(data, f, indent=2, ensure_ascii=False)
        except Exception as e:
            print(f"Failed to save config: {e}")

    def get(self, key: str, default: Any = None) -> Any:
        """获取配置值"""
        return getattr(self.config, key, default)

    def set(self, key: str, value: Any) -> None:
        """设置配置值"""
        if hasattr(self.config, key):
            setattr(self.config, key, value)
            self.save()

    def add_recent_file(self, file_path: str) -> None:
        """添加最近文件"""
        if file_path in self.config.recent_files:
            self.config.recent_files.remove(file_path)
        self.config.recent_files.insert(0, file_path)
        if len(self.config.recent_files) > self.config.max_recent_files:
            self.config.recent_files = self.config.recent_files[:self.config.max_recent_files]
        self.save()
```
