## 2. DXF解析模块

### 2.1 模块职责
将 DXF 文件解析为内部 `Contour` 数据结构，包括：
- 提取图形实体（LWPOLYLINE, POLYLINE, LINE, SPLINE, ARC, CIRCLE）
- 识别路径颜色并分类（普通轮廓、连襟框、底孔、面板花纹）
- 建立外轮廓与内孔的包含关系

### 2.2 类设计

```python
class DXFParser:
    """DXF文件解析器"""

    # 标准RGB颜色映射
    COLOR_MAP = {
        (0, 255, 255): PathType.CONNECTED,   # 青色 → 连襟框
        (0, 255, 0): PathType.HOLE,          # 绿色 → 底孔
        (0, 0, 255): PathType.PATTERN,       # 蓝色 → 面板花纹
    }

    # DXF颜色索引到RGB的映射（部分）
    ACI_TO_RGB = {
        1: (255, 0, 0),      # 红
        2: (255, 255, 0),    # 黄
        3: (0, 255, 0),      # 绿
        4: (0, 255, 255),    # 青
        5: (0, 0, 255),      # 蓝
        6: (255, 0, 255),    # 紫
        7: (255, 255, 255),  # 白
        # ...完整256色表
    }

    def parse(self, file_path: str) -> List[Contour]:
        """
        解析DXF文件

        Returns:
            轮廓列表，每个轮廓包含外边界和内孔
        """
        doc = ezdxf.readfile(file_path)
        msp = doc.modelspace()

        # 1. 提取所有路径实体
        paths = self._extract_all_paths(msp)

        # 2. 分类路径（外轮廓、内孔、特殊路径）
        contours = self._classify_and_group_paths(paths)

        return contours

    def _extract_all_paths(self, msp) -> List[Path]:
        """提取模型空间中的所有路径"""
        paths = []

        for entity in msp:
            if entity.dxftype() == 'LWPOLYLINE':
                path = self._parse_lwpolyline(entity)
            elif entity.dxftype() == 'POLYLINE':
                path = self._parse_polyline(entity)
            elif entity.dxftype() == 'LINE':
                path = self._parse_line(entity)
            elif entity.dxftype() == 'SPLINE':
                path = self._parse_spline(entity)
            elif entity.dxftype() == 'ARC':
                path = self._parse_arc(entity)
            elif entity.dxftype() == 'CIRCLE':
                path = self._parse_circle(entity)
            else:
                continue

            if path and len(path.points) >= 3:
                paths.append(path)

        return paths

    def _parse_lwpolyline(self, entity) -> Path:
        """解析轻量多段线"""
        points = []
        for x, y, *_ in entity.get_points():
            points.append(Point(x, y))

        color = self._get_entity_color(entity)
        path_type = self._determine_path_type(color)

        return Path(
            points=points,
            path_type=path_type,
            color=color
        )

    def _parse_spline(self, entity, resolution: int = 50) -> Path:
        """
        解析样条曲线

        将样条曲线离散化为多边形
        """
        points = []
        try:
            # 使用ezdxf的样条曲线采样
            for point in entity.flattening(resolution):
                points.append(Point(point.x, point.y))
        except:
            # 回退方案：使用控制点
            for cp in entity.control_points:
                points.append(Point(cp.x, cp.y))

        color = self._get_entity_color(entity)
        return Path(points=points, color=color)

    def _parse_arc(self, entity, resolution: int = 32) -> Path:
        """
        解析圆弧

        将圆弧离散化为多边形点
        """
        import math

        center = entity.dxf.center
        radius = entity.dxf.radius
        start_angle = math.radians(entity.dxf.start_angle)
        end_angle = math.radians(entity.dxf.end_angle)

        # 处理角度跨越0度的情况
        if end_angle < start_angle:
            end_angle += 2 * math.pi

        points = []
        angle_step = (end_angle - start_angle) / resolution

        for i in range(resolution + 1):
            angle = start_angle + i * angle_step
            x = center.x + radius * math.cos(angle)
            y = center.y + radius * math.sin(angle)
            points.append(Point(x, y))

        color = self._get_entity_color(entity)
        return Path(points=points, color=color)

    def _parse_circle(self, entity, resolution: int = 64) -> Path:
        """解析圆形"""
        import math

        center = entity.dxf.center
        radius = entity.dxf.radius

        points = []
        for i in range(resolution):
            angle = 2 * math.pi * i / resolution
            x = center.x + radius * math.cos(angle)
            y = center.y + radius * math.sin(angle)
            points.append(Point(x, y))

        color = self._get_entity_color(entity)
        return Path(points=points, color=color)

    def _get_entity_color(self, entity) -> Tuple[int, int, int]:
        """获取实体的RGB颜色"""
        # 优先使用真彩色
        if hasattr(entity.dxf, 'true_color') and entity.dxf.true_color:
            tc = entity.dxf.true_color
            return ((tc >> 16) & 0xFF, (tc >> 8) & 0xFF, tc & 0xFF)

        # 使用ACI颜色索引
        color_index = entity.dxf.color
        if color_index in self.ACI_TO_RGB:
            return self.ACI_TO_RGB[color_index]

        return (0, 0, 0)  # 默认黑色

    def _determine_path_type(self, color: Tuple[int, int, int]) -> PathType:
        """根据颜色确定路径类型"""
        # 使用颜色容差匹配
        for target_color, path_type in self.COLOR_MAP.items():
            if self._colors_similar(color, target_color, tolerance=30):
                return path_type
        return PathType.OUTER  # 默认为外轮廓

    def _colors_similar(self, c1, c2, tolerance: int = 30) -> bool:
        """判断两个颜色是否相近"""
        return all(abs(a - b) <= tolerance for a, b in zip(c1, c2))

    def _classify_and_group_paths(self, paths: List[Path]) -> List[Contour]:
        """
        分类并分组路径

        算法：
        1. 分离特殊路径（连襟框、底孔、花纹）
        2. 对普通路径按面积排序
        3. 使用包含关系判断内外轮廓
        4. 建立轮廓-内孔映射
        """
        # 分离特殊路径
        special_paths = {
            PathType.CONNECTED: [],
            PathType.HOLE: [],
            PathType.PATTERN: [],
        }
        normal_paths = []

        for path in paths:
            if path.path_type in special_paths:
                special_paths[path.path_type].append(path)
            else:
                normal_paths.append(path)

        # 按面积降序排序（外轮廓通常面积最大）
        normal_paths.sort(key=lambda p: self._calculate_area(p), reverse=True)

        # 建立包含关系
        contours = []
        used = set()

        for i, path in enumerate(normal_paths):
            if i in used:
                continue

            # 这是一个外轮廓
            outer = path
            outer.path_type = PathType.OUTER
            holes = []

            # 查找包含在此轮廓内的路径作为内孔
            for j, inner_path in enumerate(normal_paths):
                if j in used or j == i:
                    continue

                if self._path_contains_path(outer, inner_path):
                    # 检查是否被其他更小的轮廓包含
                    is_directly_inside = True
                    for k, middle_path in enumerate(normal_paths):
                        if k in used or k == i or k == j:
                            continue
                        if (self._path_contains_path(outer, middle_path) and
                            self._path_contains_path(middle_path, inner_path)):
                            is_directly_inside = False
                            break

                    if is_directly_inside:
                        inner_path.path_type = PathType.INNER
                        holes.append(inner_path)
                        used.add(j)

            # 创建轮廓对象
            contour = Contour(
                outer=outer,
                holes=holes,
                connected_frames=special_paths[PathType.CONNECTED],
                patterns=special_paths[PathType.PATTERN]
            )
            contours.append(contour)
            used.add(i)

        return contours

    def _calculate_area(self, path: Path) -> float:
        """
        计算多边形面积（使用Shoelace公式）

        正值表示逆时针，负值表示顺时针
        """
        n = len(path.points)
        if n < 3:
            return 0

        area = 0
        for i in range(n):
            j = (i + 1) % n
            area += path.points[i].x * path.points[j].y
            area -= path.points[j].x * path.points[i].y

        return abs(area) / 2

    def _path_contains_path(self, outer: Path, inner: Path) -> bool:
        """
        判断outer路径是否包含inner路径

        使用inner的第一个点进行点-多边形测试
        """
        if not inner.points:
            return False
        test_point = inner.points[0]
        return self._point_in_polygon(test_point, outer)

    def _point_in_polygon(self, point: Point, polygon: Path) -> bool:
        """
        射线法判断点是否在多边形内

        从点向右发射射线，计算与多边形边的交点数
        奇数交点表示在内部
        """
        x, y = point.x, point.y
        n = len(polygon.points)
        inside = False

        j = n - 1
        for i in range(n):
            xi, yi = polygon.points[i].x, polygon.points[i].y
            xj, yj = polygon.points[j].x, polygon.points[j].y

            if ((yi > y) != (yj > y)) and (x < (xj - xi) * (y - yi) / (yj - yi) + xi):
                inside = not inside
            j = i

        return inside
```

## 3. 路径偏移模块

### 3.1 模块职责
使用 pyclipper 库进行路径的内/外偏移操作，支持布尔运算。

### 3.2 类设计

```python
import pyclipper

class JoinType(Enum):
    """拐角连接类型"""
    MITER = pyclipper.JT_MITER   # 尖角
    ROUND = pyclipper.JT_ROUND   # 圆角
    SQUARE = pyclipper.JT_SQUARE # 方角

class EndType(Enum):
    """端点类型"""
    CLOSED_POLYGON = pyclipper.ET_CLOSEDPOLYGON
    CLOSED_LINE = pyclipper.ET_CLOSEDLINE
    OPEN_ROUND = pyclipper.ET_OPENROUND
    OPEN_SQUARE = pyclipper.ET_OPENSQUARE
    OPEN_BUTT = pyclipper.ET_OPENBUTT

class PathOffset:
    """路径偏移处理器"""

    SCALE = 1000  # 缩放因子（pyclipper使用整数运算）
    MITER_LIMIT = 2.0  # 尖角限制

    def offset(self, path: Path, distance: float,
               join_type: JoinType = JoinType.MITER) -> List[Path]:
        """
        对单个路径进行偏移

        Args:
            path: 输入路径
            distance: 偏移距离（正=外扩，负=内缩）
            join_type: 拐角处理方式

        Returns:
            偏移后的路径列表（可能产生多个或为空）
        """
        # 转换为整数坐标
        scaled_path = self._scale_path(path)

        # 创建偏移器
        pco = pyclipper.PyclipperOffset()
        pco.MiterLimit = self.MITER_LIMIT

        # 添加路径
        pco.AddPath(scaled_path, join_type.value,
                    EndType.CLOSED_POLYGON.value)

        # 执行偏移
        scaled_distance = distance * self.SCALE
        solution = pco.Execute(scaled_distance)

        # 转换回浮点坐标
        result_paths = []
        for scaled_result in solution:
            points = [Point(p[0] / self.SCALE, p[1] / self.SCALE)
                      for p in scaled_result]
            result_paths.append(Path(points=points, path_type=path.path_type))

        return result_paths

    def offset_contour(self, contour: Contour, distance: float,
                        join_type: JoinType = JoinType.MITER) -> Contour:
        """
        对整个轮廓进行偏移

        外轮廓和内孔的偏移方向相反：
        - 外扩时：外轮廓向外，内孔向内
        - 内缩时：外轮廓向内，内孔向外
        """
        # 偏移外轮廓
        outer_results = self.offset(contour.outer, distance, join_type)
        if not outer_results:
            return None

        new_outer = outer_results[0]

        # 偏移内孔（方向相反）
        new_holes = []
        for hole in contour.holes:
            hole_results = self.offset(hole, -distance, join_type)
            new_holes.extend(hole_results)

        return Contour(
            outer=new_outer,
            holes=new_holes,
            connected_frames=contour.connected_frames,
            patterns=contour.patterns,
            name=contour.name
        )

    def multi_offset(self, path: Path, distances: List[float],
                      join_type: JoinType = JoinType.MITER) -> List[List[Path]]:
        """
        对路径进行多次偏移（用于多壁生成）

        Args:
            path: 输入路径
            distances: 偏移距离列表

        Returns:
            每个偏移距离对应的路径列表
        """
        results = []
        for distance in distances:
            offset_paths = self.offset(path, distance, join_type)
            results.append(offset_paths)
        return results

    def boolean_union(self, paths: List[Path]) -> List[Path]:
        """
        布尔并集运算

        将多个路径合并为一个或多个不重叠的路径
        """
        if not paths:
            return []

        pc = pyclipper.Pyclipper()

        for i, path in enumerate(paths):
            scaled = self._scale_path(path)
            if i == 0:
                pc.AddPath(scaled, pyclipper.PT_SUBJECT, True)
            else:
                pc.AddPath(scaled, pyclipper.PT_CLIP, True)

        solution = pc.Execute(pyclipper.CT_UNION,
                               pyclipper.PFT_NONZERO,
                               pyclipper.PFT_NONZERO)

        return self._unscale_paths(solution)

    def boolean_difference(self, subject: Path, clip: Path) -> List[Path]:
        """
        布尔差集运算

        从subject中减去clip区域
        """
        pc = pyclipper.Pyclipper()

        pc.AddPath(self._scale_path(subject), pyclipper.PT_SUBJECT, True)
        pc.AddPath(self._scale_path(clip), pyclipper.PT_CLIP, True)

        solution = pc.Execute(pyclipper.CT_DIFFERENCE,
                               pyclipper.PFT_NONZERO,
                               pyclipper.PFT_NONZERO)

        return self._unscale_paths(solution)

    def boolean_intersection(self, subject: Path, clip: Path) -> List[Path]:
        """
        布尔交集运算

        返回subject和clip的重叠区域
        """
        pc = pyclipper.Pyclipper()

        pc.AddPath(self._scale_path(subject), pyclipper.PT_SUBJECT, True)
        pc.AddPath(self._scale_path(clip), pyclipper.PT_CLIP, True)

        solution = pc.Execute(pyclipper.CT_INTERSECTION,
                               pyclipper.PFT_NONZERO,
                               pyclipper.PFT_NONZERO)

        return self._unscale_paths(solution)

    def _scale_path(self, path: Path) -> List[Tuple[int, int]]:
        """将路径坐标缩放为整数"""
        return [(int(p.x * self.SCALE), int(p.y * self.SCALE))
                for p in path.points]

    def _unscale_paths(self, scaled_paths: List) -> List[Path]:
        """将整数坐标转换回浮点路径"""
        result = []
        for scaled in scaled_paths:
            points = [Point(p[0] / self.SCALE, p[1] / self.SCALE)
                      for p in scaled]
            result.append(Path(points=points))
        return result

    def simplify_path(self, path: Path, tolerance: float = 0.1) -> Path:
        """
        简化路径（Douglas-Peucker算法）

        减少路径点数，保持形状
        """
        scaled = self._scale_path(path)
        simplified = pyclipper.SimplifyPolygon(
            scaled,
            pyclipper.PFT_NONZERO
        )

        if simplified:
            points = [Point(p[0] / self.SCALE, p[1] / self.SCALE)
                      for p in simplified[0]]
            return Path(points=points, path_type=path.path_type)
        return path

    def clean_path(self, path: Path, distance: float = 0.1) -> Path:
        """
        清理路径（移除过近的点）
        """
        scaled = self._scale_path(path)
        cleaned = pyclipper.CleanPolygon(scaled, distance * self.SCALE)

        points = [Point(p[0] / self.SCALE, p[1] / self.SCALE)
                  for p in cleaned]
        return Path(points=points, path_type=path.path_type)
```

## 4. 切片引擎模块

### 4.1 模块职责
核心切片逻辑，协调各子生成器，将轮廓数据转换为打印层数据。

### 4.2 类设计

```python
@dataclass
class LayerInfo:
    """层信息（用于规划）"""
    z_height: float          # 层底部高度
    layer_height: float      # 层厚度
    extruder: ExtruderType   # 使用的挤出机
    is_first_layer: bool     # 是否首层
    is_step_layer: bool      # 是否台阶层
    step_config: Optional[StepConfig] = None  # 台阶配置
    has_skirt: bool = False  # 是否有防翘裙边
    has_edge_wrap: bool = False  # 是否有包边/勾边


class SliceEngine:
    """切片引擎"""

    def __init__(self):
        self.path_offset = PathOffset()
        self.bevel_generator = BevelGenerator(self.path_offset)
        self.step_generator = StepGenerator(self.path_offset)
        self.fill_generator = FillGenerator(self.path_offset)
        self.skirt_generator = SkirtGenerator()
        self.connected_generator = ConnectedGenerator(self.path_offset)

    def slice(self, contours: List[Contour], params: SliceParams) -> SliceResult:
        """
        执行切片操作

        主流程：
        1. 计算层规划
        2. 逐层生成路径
        3. 处理特殊结构
        4. 计算统计信息
        """
        # 1. 计算层规划
        layer_infos = self._plan_layers(params)

        # 2. 逐层生成
        layers = []
        for info in layer_infos:
            layer = self._generate_layer(contours, info, params)
            layers.append(layer)

        # 3. 添加预打边线（安全线）
        if params.shell.safety_line_count > 0:
            safety_layers = self._generate_safety_lines(contours, params)
            # 安全线在最底层之前打印
            layers = safety_layers + layers

        # 4. 计算统计
        result = self._create_result(layers, params)

        return result

    def _plan_layers(self, params: SliceParams) -> List[LayerInfo]:
        """
        规划所有层

        考虑因素：
        - 总高度和层高
        - 换色层配置
        - 台阶位置
        - 防翘裙边位置
        - 包边/勾边位置
        """
        layer_height = params.shell.layer_height
        total_height = params.shell.height
        layer_count = int(total_height / layer_height)

        # 构建换色层映射
        color_layers = self._build_color_layer_map(params.color, total_height)

        # 构建台阶层映射
        step_layers = self._build_step_layer_map(params.step, layer_height)

        layer_infos = []
        for i in range(layer_count):
            z = i * layer_height
            z_top = z + layer_height

            # 确定挤出机
            extruder = self._get_extruder_at_height(z, color_layers)

            # 检查是否台阶层
            is_step = z in step_layers
            step_config = step_layers.get(z)

            # 检查是否首层（需要防翘裙边）
            is_first = (i == 0)
            has_skirt = is_first and params.skirt.enabled

            # 检查是否顶层（需要包边/勾边）
            is_last = (i == layer_count - 1)
            has_edge_wrap = is_last and (params.edge_wrap.wrap_enabled or
                                          params.edge_wrap.outline_enabled)

            info = LayerInfo(
                z_height=z,
                layer_height=layer_height,
                extruder=extruder,
                is_first_layer=is_first,
                is_step_layer=is_step,
                step_config=step_config,
                has_skirt=has_skirt,
                has_edge_wrap=has_edge_wrap
            )
            layer_infos.append(info)

        return layer_infos

    def _build_color_layer_map(self, color_params: ColorParams,
                                 total_height: float) -> Dict[float, ExtruderType]:
        """构建换色层高度-挤出机映射"""
        if not color_params.enabled or not color_params.layers:
            return {}

        mapping = {}
        current_height = 0

        for config in color_params.layers:
            end_height = current_height + config.height
            # 将该高度范围内的所有层映射到指定挤出机
            mapping[current_height] = (end_height, config.extruder)
            current_height = end_height

        return mapping

    def _build_step_layer_map(self, step_params: StepParams,
                                layer_height: float) -> Dict[float, StepConfig]:
        """构建台阶层高度-配置映射"""
        mapping = {}
        for step in step_params.steps:
            if step.enabled:
                # 对齐到层高
                aligned_z = round(step.position / layer_height) * layer_height
                mapping[aligned_z] = step
        return mapping

    def _get_extruder_at_height(self, z: float,
                                  color_layers: Dict) -> ExtruderType:
        """获取指定高度应使用的挤出机"""
        for start_z, (end_z, extruder) in color_layers.items():
            if start_z <= z < end_z:
                return extruder
        return ExtruderType.EXTRUDER_1

    def _generate_layer(self, contours: List[Contour],
                         info: LayerInfo,
                         params: SliceParams) -> Layer:
        """
        生成单层

        处理顺序：
        1. 应用边缘轮廓（斜边收缩）
        2. 生成壁路径
        3. 生成台阶（如需要）
        4. 生成防翘裙边（如需要）
        5. 生成填充（如需要）
        6. 生成包边/勾边（如需要）
        """
        layer = Layer(
            z_height=info.z_height,
            layer_height=info.layer_height,
            extruder=info.extruder
        )

        for contour in contours:
            # 1. 应用边缘轮廓
            modified_contour = self.bevel_generator.apply_edge_profile(
                contour, info.z_height, params.shell
            )
            if not modified_contour:
                continue

            # 2. 生成壁路径
            wall_paths = self._generate_wall_paths(
                modified_contour, info, params.shell
            )
            layer.paths.extend(wall_paths)

            # 3. 生成台阶
            if info.is_step_layer and info.step_config:
                step_paths = self.step_generator.generate_steps(
                    modified_contour, info.step_config, params.step
                )
                layer.paths.extend(step_paths)

            # 4. 生成防翘裙边
            if info.has_skirt:
                skirt_paths = self.skirt_generator.generate_skirt(
                    modified_contour, params.skirt
                )
                layer.paths.extend(skirt_paths)

            # 5. 生成填充
            if info.is_first_layer and params.baseplate.enabled:
                fill_paths = self.fill_generator.generate_baseplate_fill(
                    modified_contour, params.baseplate
                )
                layer.paths.extend(fill_paths)

        # 6. 处理连襟字
        if params.connected.enabled:
            connected_paths = self.connected_generator.generate_connected(
                contours, info.z_height, params.connected
            )
            layer.paths.extend(connected_paths)

        return layer

    def _generate_wall_paths(self, contour: Contour,
                               info: LayerInfo,
                               shell_params: ShellParams) -> List[LayerPath]:
        """
        生成壁路径

        支持多层壁（走线次数）
        """
        paths = []
        wall_thickness = shell_params.wall_thickness
        wall_count = shell_params.wall_count
        wall_offset = shell_params.wall_offset

        for i in range(wall_count):
            # 计算当前壁的偏移距离
            offset_distance = -(wall_thickness / 2 + i * wall_offset)

            # 偏移外轮廓
            offset_results = self.path_offset.offset(
                contour.outer, offset_distance
            )

            for offset_path in offset_results:
                layer_path = LayerPath(
                    points=offset_path.points,
                    role=PathRole.WALL,
                    extruder=info.extruder,
                    line_width=wall_thickness
                )
                paths.append(layer_path)

            # 偏移内孔（方向相反）
            for hole in contour.holes:
                hole_offset_results = self.path_offset.offset(
                    hole, -offset_distance
                )
                for offset_hole in hole_offset_results:
                    layer_path = LayerPath(
                        points=offset_hole.points,
                        role=PathRole.WALL,
                        extruder=info.extruder,
                        line_width=wall_thickness
                    )
                    paths.append(layer_path)

        return paths

    def _generate_safety_lines(self, contours: List[Contour],
                                  params: SliceParams) -> List[Layer]:
        """
        生成预打边线（安全线）

        围绕所有字形外围生成边线，防止首层打印时材料粘附不良
        """
        layers = []
        offset = params.shell.safety_line_offset
        count = params.shell.safety_line_count

        for i in range(count):
            layer = Layer(
                z_height=0,
                layer_height=params.shell.layer_height,
                extruder=ExtruderType.EXTRUDER_1
            )

            for contour in contours:
                # 向外偏移
                offset_distance = offset + i * params.shell.wall_thickness
                offset_paths = self.path_offset.offset(
                    contour.outer, offset_distance
                )

                for path in offset_paths:
                    layer_path = LayerPath(
                        points=path.points,
                        role=PathRole.SAFETY_LINE,
                        extruder=ExtruderType.EXTRUDER_1,
                        line_width=params.shell.wall_thickness
                    )
                    layer.paths.append(layer_path)

            layers.append(layer)

        return layers

    def _create_result(self, layers: List[Layer],
                        params: SliceParams) -> SliceResult:
        """创建切片结果，计算统计信息"""
        total_distance = 0
        filament_per_extruder = {
            ExtruderType.EXTRUDER_1: 0,
            ExtruderType.EXTRUDER_2: 0,
            ExtruderType.EXTRUDER_3: 0,
        }

        for layer in layers:
            for path in layer.paths:
                # 计算路径长度
                distance = self._calculate_path_length(path.points)
                total_distance += distance

                # 按挤出机统计
                filament_per_extruder[path.extruder] += distance

        # 估算时间（基于打印速度）
        avg_speed = 40  # mm/s, 平均速度
        estimated_time = total_distance / avg_speed

        # 将距离转换为线材长度（考虑截面积）
        line_width = params.shell.wall_thickness
        layer_height = params.shell.layer_height
        filament_diameter = 1.75  # mm

        cross_section_ratio = (line_width * layer_height) / (
            3.14159 * (filament_diameter / 2) ** 2
        )

        estimated_filament = total_distance * cross_section_ratio / 1000  # 转换为米

        return SliceResult(
            layers=layers,
            total_layers=len(layers),
            total_height=params.shell.height,
            estimated_time=estimated_time,
            estimated_filament=estimated_filament,
            filament_per_extruder={
                k: v * cross_section_ratio / 1000
                for k, v in filament_per_extruder.items()
            }
        )

    def _calculate_path_length(self, points: List[Point]) -> float:
        """计算路径总长度"""
        import math
        length = 0
        for i in range(len(points)):
            j = (i + 1) % len(points)
            dx = points[j].x - points[i].x
            dy = points[j].y - points[i].y
            length += math.sqrt(dx * dx + dy * dy)
        return length
```

```python
class BevelGenerator:
    """边缘生成器"""

    def __init__(self, path_offset: PathOffset):
        self.path_offset = path_offset

    def apply_edge_profile(self, contour: Contour, z: float,
                            params: ShellParams) -> Optional[Contour]:
        """
        根据边缘类型和高度应用轮廓变换

        Args:
            contour: 原始轮廓
            z: 当前层高度
            params: 字壳参数

        Returns:
            变换后的轮廓（可能缩小到消失返回None）
        """
        if params.edge_type == EdgeType.STRAIGHT:
            return self._apply_straight_edge(contour, z, params)
        elif params.edge_type == EdgeType.BEVELED:
            return self._apply_beveled_edge(contour, z, params)
        elif params.edge_type == EdgeType.CUSTOM:
            return self._apply_custom_edge(contour, z, params)
        else:
            return contour

    def _apply_straight_edge(self, contour: Contour, z: float,
                               params: ShellParams) -> Contour:
        """
        直边处理

        直边字壳从底到顶保持相同尺寸
        """
        # 直边不需要偏移
        return contour

    def _apply_beveled_edge(self, contour: Contour, z: float,
                              params: ShellParams) -> Optional[Contour]:
        """
        斜边处理

        斜边结构：
        ┌─────────┐ ← 面板位置（最内收缩）
        │ ╲     ╱ │
        │  ╲   ╱  │ ← 斜边区域（线性收缩）
        │   ╲ ╱   │
        │    │    │ ← 底板区域（不收缩）
        └────┴────┘

        收缩方向：从外向内
        """
        baseplate_top = params.baseplate_thickness
        bevel_end = baseplate_top + params.bevel_width
        max_offset = params.bevel_width  # 最大内缩距离

        if z < baseplate_top:
            # 底板区域：不收缩
            offset = 0
        elif z < bevel_end:
            # 斜边区域：线性收缩
            progress = (z - baseplate_top) / params.bevel_width
            offset = -progress * max_offset
        else:
            # 面板区域：保持最大收缩
            offset = -max_offset

        if abs(offset) < 0.001:
            return contour

        # 应用偏移
        return self.path_offset.offset_contour(contour, offset)

    def _apply_custom_edge(self, contour: Contour, z: float,
                             params: ShellParams) -> Optional[Contour]:
        """
        异形边处理

        根据EdgeProfile定义的点序列进行插值
        """
        if not params.edge_profile:
            return contour

        profile = params.edge_profile
        total_height = params.height

        # 归一化高度
        normalized_z = z / total_height

        # 在profile中查找对应的偏移量
        offset = self._interpolate_profile(profile, normalized_z)

        if abs(offset) < 0.001:
            return contour

        return self.path_offset.offset_contour(contour, offset)

    def _interpolate_profile(self, profile: EdgeProfile,
                               normalized_z: float) -> float:
        """
        在异形边轮廓中插值

        profile.points 是 [(normalized_height, offset), ...] 的列表
        """
        points = profile.points
        if not points:
            return 0

        # 找到z所在的区间
        for i in range(len(points) - 1):
            z1, offset1 = points[i].x, points[i].y
            z2, offset2 = points[i + 1].x, points[i + 1].y

            if z1 <= normalized_z <= z2:
                # 线性插值
                t = (normalized_z - z1) / (z2 - z1) if z2 != z1 else 0
                return offset1 + t * (offset2 - offset1)

        # 超出范围，使用边界值
        if normalized_z < points[0].x:
            return points[0].y
        return points[-1].y

    def get_preview_profile(self, params: ShellParams) -> List[Point]:
        """
        获取边缘预览轮廓（用于UI显示）

        返回边缘截面的2D点序列
        """
        height = params.height
        steps = 50
        points = []

        for i in range(steps + 1):
            z = height * i / steps
            # 模拟一个单位宽度的轮廓
            if params.edge_type == EdgeType.STRAIGHT:
                offset = 0
            elif params.edge_type == EdgeType.BEVELED:
                baseplate_top = params.baseplate_thickness
                bevel_end = baseplate_top + params.bevel_width

                if z < baseplate_top:
                    offset = 0
                elif z < bevel_end:
                    progress = (z - baseplate_top) / params.bevel_width
                    offset = progress * params.bevel_width
                else:
                    offset = params.bevel_width
            else:
                offset = 0

            points.append(Point(offset, z))

        return points
```

## 6. 填充生成模块

### 6.1 模块职责
生成各类填充图案：面板填充、网格填充、底板填充。

### 6.2 类设计

```python
class FillGenerator:
    """填充图案生成器"""

    def __init__(self, path_offset: PathOffset):
        self.path_offset = path_offset

    def generate_panel_fill(self, contour: Contour,
                             params: PanelParams) -> List[LayerPath]:
        """
        生成面板填充

        面板填充结构：
        1. 正面（1类填充）：较大间距，形成孔洞
        2. 背面（2类填充）：较小间距，遮光
        """
        paths = []

        # 内缩边界
        fill_contour = self.path_offset.offset_contour(
            contour, -params.fill_config.boundary_distance
        )
        if not fill_contour:
            return paths

        # 正面填充
        front_paths = self._generate_line_fill(
            fill_contour,
            params.fill_config.spacing,
            params.fill_config.angle,
            params.front_layer.extrusion_width
        )
        for path in front_paths:
            path.role = PathRole.FILL_PANEL
            path.extruder = params.front_layer.extruder or ExtruderType.EXTRUDER_1
        paths.extend(front_paths)

        # 背面填充（交叉角度）
        back_angle = params.fill_config.angle + 90
        back_paths = self._generate_line_fill(
            fill_contour,
            params.fill_config.spacing2,
            back_angle,
            params.back_layer.extrusion_width
        )
        for path in back_paths:
            path.role = PathRole.FILL_PANEL
            path.extruder = params.back_layer.extruder or ExtruderType.EXTRUDER_1
        paths.extend(back_paths)

        return paths

    def generate_grid_fill(self, contour: Contour,
                            params: GridParams) -> List[LayerPath]:
        """
        生成网格填充

        网格用于灌胶工艺，需要形成规则的方格
        """
        paths = []

        fill_contour = self.path_offset.offset_contour(
            contour, -params.fill_config.boundary_distance
        )
        if not fill_contour:
            return paths

        # 生成水平线
        h_paths = self._generate_line_fill(
            fill_contour,
            params.fill_config.spacing,
            0,  # 水平
            params.layer.extrusion_width
        )

        # 生成垂直线
        v_paths = self._generate_line_fill(
            fill_contour,
            params.fill_config.spacing,
            90,  # 垂直
            params.layer.extrusion_width
        )

        for path in h_paths + v_paths:
            path.role = PathRole.FILL_GRID
        paths.extend(h_paths)
        paths.extend(v_paths)

        return paths

    def generate_baseplate_fill(self, contour: Contour,
                                  params: BaseplateParams) -> List[LayerPath]:
        """
        生成底板填充

        底板填充需要密实，防止漏光
        """
        paths = []

        fill_contour = self.path_offset.offset_contour(
            contour, -params.fill_config.boundary_distance
        )
        if not fill_contour:
            return paths

        # 密实填充
        fill_paths = self._generate_line_fill(
            fill_contour,
            params.fill_config.spacing,
            params.fill_config.angle,
            params.layer.extrusion_width
        )

        for path in fill_paths:
            path.role = PathRole.FILL_BASEPLATE
        paths.extend(fill_paths)

        return paths

    def _generate_line_fill(self, contour: Contour, spacing: float,
                             angle: float, line_width: float) -> List[LayerPath]:
        """
        生成线性填充

        算法：
        1. 旋转轮廓到水平方向
        2. 生成等间距水平扫描线
        3. 与轮廓求交
        4. 连接相邻线段（减少空走）
        5. 反向旋转恢复
        """
        import math

        # 获取边界框
        min_x, min_y, max_x, max_y = contour.outer.bounds

        # 扩展边界（确保覆盖旋转后的区域）
        diagonal = math.sqrt((max_x - min_x) ** 2 + (max_y - min_y) ** 2)
        center_x = (min_x + max_x) / 2
        center_y = (min_y + max_y) / 2

        # 旋转角度（弧度）
        rad = math.radians(angle)
        cos_a = math.cos(rad)
        sin_a = math.sin(rad)

        # 生成扫描线
        scan_lines = []
        y = center_y - diagonal / 2
        while y < center_y + diagonal / 2:
            # 旋转后的扫描线端点
            x1 = center_x - diagonal
            x2 = center_x + diagonal

            # 旋转回原坐标系
            start = Point(
                center_x + (x1 - center_x) * cos_a - (y - center_y) * sin_a,
                center_y + (x1 - center_x) * sin_a + (y - center_y) * cos_a
            )
            end = Point(
                center_x + (x2 - center_x) * cos_a - (y - center_y) * sin_a,
                center_y + (x2 - center_x) * sin_a + (y - center_y) * cos_a
            )

            scan_lines.append((start, end))
            y += spacing

        # 与轮廓求交
        clipped_segments = self._clip_lines_to_contour(scan_lines, contour)

        # 连接相邻线段
        connected_paths = self._connect_segments(clipped_segments)

        # 创建LayerPath对象
        paths = []
        for segment_points in connected_paths:
            if len(segment_points) >= 2:
                path = LayerPath(
                    points=segment_points,
                    role=PathRole.FILL_PANEL,
                    line_width=line_width,
                    is_closed=False
                )
                paths.append(path)

        return paths

    def _clip_lines_to_contour(self, lines: List[Tuple[Point, Point]],
                                 contour: Contour) -> List[Tuple[Point, Point]]:
        """
        裁剪线段到轮廓内部

        使用线段-多边形求交算法
        """
        clipped = []

        for start, end in lines:
            intersections = self._line_polygon_intersections(
                start, end, contour.outer
            )

            # 减去内孔的交点
            for hole in contour.holes:
                hole_intersections = self._line_polygon_intersections(
                    start, end, hole
                )
                # 处理孔的减法（复杂，简化处理）

            # 配对交点形成线段
            if len(intersections) >= 2:
                intersections.sort(key=lambda p: (p.x - start.x) ** 2 + (p.y - start.y) ** 2)
                for i in range(0, len(intersections) - 1, 2):
                    clipped.append((intersections[i], intersections[i + 1]))

        return clipped

    def _line_polygon_intersections(self, start: Point, end: Point,
                                      polygon: Path) -> List[Point]:
        """计算线段与多边形的所有交点"""
        intersections = []
        n = len(polygon.points)

        for i in range(n):
            j = (i + 1) % n
            p1 = polygon.points[i]
            p2 = polygon.points[j]

            intersection = self._line_line_intersection(start, end, p1, p2)
            if intersection:
                intersections.append(intersection)

        return intersections

    def _line_line_intersection(self, p1: Point, p2: Point,
                                  p3: Point, p4: Point) -> Optional[Point]:
        """
        计算两线段的交点

        使用参数方程法
        """
        x1, y1 = p1.x, p1.y
        x2, y2 = p2.x, p2.y
        x3, y3 = p3.x, p3.y
        x4, y4 = p4.x, p4.y

        denom = (x1 - x2) * (y3 - y4) - (y1 - y2) * (x3 - x4)
        if abs(denom) < 1e-10:
            return None

        t = ((x1 - x3) * (y3 - y4) - (y1 - y3) * (x3 - x4)) / denom
        u = -((x1 - x2) * (y1 - y3) - (y1 - y2) * (x1 - x3)) / denom

        if 0 <= t <= 1 and 0 <= u <= 1:
            x = x1 + t * (x2 - x1)
            y = y1 + t * (y2 - y1)
            return Point(x, y)

        return None

    def _connect_segments(self, segments: List[Tuple[Point, Point]]) -> List[List[Point]]:
        """
        连接相邻线段减少空走

        使用贪心算法：每次选择最近的下一个线段
        """
        if not segments:
            return []

        # 简单实现：按Y坐标排序，交替方向连接
        sorted_segments = sorted(segments, key=lambda s: s[0].y)

        connected = []
        current_path = []
        reverse_next = False

        for seg in sorted_segments:
            start, end = seg
            if reverse_next:
                start, end = end, start

            if current_path:
                # 检查是否可以连接
                last_point = current_path[-1]
                distance = ((start.x - last_point.x) ** 2 +
                            (start.y - last_point.y) ** 2) ** 0.5

                if distance > 5:  # 超过阈值，开始新路径
                    if len(current_path) >= 2:
                        connected.append(current_path)
                    current_path = [start, end]
                else:
                    current_path.extend([start, end])
            else:
                current_path = [start, end]

            reverse_next = not reverse_next

        if len(current_path) >= 2:
            connected.append(current_path)

        return connected
```

## 7. G-code生成模块

### 7.1 模块职责
将层数据转换为机器可执行的G-code指令。

### 7.2 类设计

```python
class GCodeWriter:
    """G-code生成器"""

    def __init__(self, params: GCodeParams):
        self.params = params
        self.current_extruder = ExtruderType.EXTRUDER_1
        self.current_position = Point3D(0, 0, 0)
        self.e_position = 0.0
        self.is_retracted = False

    def write(self, result: SliceResult, output_path: str) -> None:
        """生成并写入G-code文件"""
        lines = []

        # 头部
        lines.extend(self._write_header())

        # 各层
        for i, layer in enumerate(result.layers):
            lines.append(f"; Layer {i}")
            lines.extend(self._write_layer(layer, is_first=(i == 0)))

        # 尾部
        lines.extend(self._write_footer())

        # 写入文件
        with open(output_path, 'w', encoding='utf-8') as f:
            f.write('\n'.join(lines))

    def _write_header(self) -> List[str]:
        """写入G-code头部"""
        lines = [
            "; Generated by SignMaker Slicer",
            f"; Machine: {self.params.machine_type}",
            f"; Print time: estimated",
            "",
            "; Startup sequence",
            "G28 ; Home all axes",
            f"M104 S{self.params.extruder1_temp} T0 ; Set extruder 1 temp",
        ]

        if self.params.extruder2_temp > 0:
            lines.append(f"M104 S{self.params.extruder2_temp} T1 ; Set extruder 2 temp")

        lines.extend([
            f"M109 S{self.params.extruder1_temp} T0 ; Wait for extruder 1 temp",
            "G90 ; Absolute positioning",
            "M82 ; Absolute extrusion",
            "G92 E0 ; Reset extruder",
            "",
            "; Prime nozzle",
            f"G1 E{self.params.prime_length} F300",
            "G92 E0",
            "",
        ])

        return lines

    def _write_layer(self, layer: Layer, is_first: bool = False) -> List[str]:
        """写入单层G-code"""
        lines = []
        z = layer.z_height + layer.layer_height

        # 移动到层高度
        lines.append(f"G1 Z{z:.3f} F{self.params.travel_speed * 60:.0f}")

        # 设置打印速度
        speed = (self.params.first_layer_speed if is_first
                 else self.params.print_speed)

        # 处理每条路径
        for path in layer.paths:
            # 检查是否需要换挤出机
            if path.extruder != self.current_extruder:
                lines.extend(self._write_color_change(path.extruder))

            # 写入路径
            lines.extend(self._write_path(path, z, speed))

        return lines

    def _write_path(self, path: LayerPath, z: float,
                     speed: float) -> List[str]:
        """写入路径G-code"""
        lines = []

        if not path.points:
            return lines

        # 移动到起点
        start = path.points[0]
        lines.extend(self._write_travel(start, z))

        # 取消回抽
        if self.is_retracted:
            lines.extend(self._write_unretract())

        # 打印路径
        for i in range(1, len(path.points)):
            point = path.points[i]
            lines.extend(self._write_print_move(point, z, speed, path.line_width))

        # 闭合路径
        if path.is_closed:
            lines.extend(self._write_print_move(start, z, speed, path.line_width))

        # 回抽
        lines.extend(self._write_retract())

        return lines

    def _write_travel(self, target: Point, z: float) -> List[str]:
        """写入空走移动"""
        lines = []

        # Z轴抬升
        hop_height = self._get_hop_height(target)
        if hop_height > 0:
            lines.append(f"G1 Z{z + hop_height:.3f} F{300 * 60:.0f}")

        # XY移动
        lines.append(
            f"G0 X{target.x:.3f} Y{target.y:.3f} "
            f"F{self.params.travel_speed * 60:.0f}"
        )

        # Z轴下降
        if hop_height > 0:
            lines.append(f"G1 Z{z:.3f} F{300 * 60:.0f}")

        self.current_position = Point3D(target.x, target.y, z)
        return lines

    def _write_print_move(self, target: Point, z: float,
                           speed: float, line_width: float) -> List[str]:
        """写入打印移动"""
        # 计算挤出量
        distance = self._calculate_distance(
            Point(self.current_position.x, self.current_position.y),
            target
        )
        extrusion = self._calculate_extrusion(distance, line_width)

        self.e_position += extrusion
        self.current_position = Point3D(target.x, target.y, z)

        return [
            f"G1 X{target.x:.3f} Y{target.y:.3f} "
            f"E{self.e_position:.5f} F{speed * 60:.0f}"
        ]

    def _write_retract(self) -> List[str]:
        """写入回抽"""
        if self.is_retracted:
            return []

        self.e_position -= self.params.retract_length
        self.is_retracted = True

        return [
            f"G1 E{self.e_position:.5f} F{self.params.retract_speed * 60:.0f} ; Retract"
        ]

    def _write_unretract(self) -> List[str]:
        """写入取消回抽"""
        if not self.is_retracted:
            return []

        self.e_position += self.params.retract_length
        self.is_retracted = False

        return [
            f"G1 E{self.e_position:.5f} F{self.params.retract_speed * 60:.0f} ; Unretract"
        ]

    def _write_color_change(self, new_extruder: ExtruderType) -> List[str]:
        """写入换色/换挤出机"""
        lines = [
            "",
            f"; Switch to extruder {new_extruder.value}",
        ]

        # 回抽当前挤出机
        lines.append(
            f"G1 E{self.e_position - self.params.color_change_retract:.5f} "
            f"F{3000:.0f} ; Long retract"
        )

        # 切换挤出机
        lines.append(f"T{new_extruder.value - 1}")

        # 补料
        self.e_position = self.params.color_change_prime
        lines.append(
            f"G1 E{self.e_position:.5f} F{3000:.0f} ; Prime"
        )

        self.current_extruder = new_extruder
        self.is_retracted = False

        lines.append("")
        return lines

    def _write_footer(self) -> List[str]:
        """写入G-code尾部"""
        return [
            "",
            "; End sequence",
            f"G1 Z{self.params.end_z_position:.0f} F{300 * 60:.0f} ; Raise Z",
            "G28 X Y ; Home X and Y",
            "M104 S0 ; Turn off extruder 1",
            "M104 S0 T1 ; Turn off extruder 2",
            "M84 ; Disable motors",
            "",
            "; End of file",
        ]

    def _calculate_distance(self, p1: Point, p2: Point) -> float:
        """计算两点间距离"""
        import math
        return math.sqrt((p2.x - p1.x) ** 2 + (p2.y - p1.y) ** 2)

    def _calculate_extrusion(self, distance: float, line_width: float) -> float:
        """
        计算挤出量

        E = (距离 × 线宽 × 层高) / (π × (线材直径/2)²)
        """
        import math

        layer_height = 0.3  # 应该从参数获取
        filament_area = math.pi * (self.params.filament_diameter / 2) ** 2
        extrusion_area = line_width * layer_height

        return distance * extrusion_area / filament_area

    def _get_hop_height(self, target: Point) -> float:
        """根据移动距离决定抬升高度"""
        distance = self._calculate_distance(
            Point(self.current_position.x, self.current_position.y),
            target
        )

        if distance > self.params.distance_threshold:
            return self.params.z_hop_height2
        elif distance > 10:
            return self.params.z_hop_height1
        return 0
```

```python
# src/utils/geometry.py

import math
from typing import List, Tuple, Optional
from src.core.types import Point, Path

def distance(p1: Point, p2: Point) -> float:
    """计算两点间距离"""
    return math.sqrt((p2.x - p1.x) ** 2 + (p2.y - p1.y) ** 2)

def path_length(path: Path) -> float:
    """计算路径总长度"""
    total = 0
    for i in range(len(path.points)):
        j = (i + 1) % len(path.points)
        total += distance(path.points[i], path.points[j])
    return total

def path_area(path: Path) -> float:
    """计算路径围成的面积（Shoelace公式）"""
    n = len(path.points)
    if n < 3:
        return 0

    area = 0
    for i in range(n):
        j = (i + 1) % n
        area += path.points[i].x * path.points[j].y
        area -= path.points[j].x * path.points[i].y

    return abs(area) / 2

def path_centroid(path: Path) -> Point:
    """计算路径质心"""
    n = len(path.points)
    if n == 0:
        return Point(0, 0)

    cx = sum(p.x for p in path.points) / n
    cy = sum(p.y for p in path.points) / n
    return Point(cx, cy)

def path_bounds(path: Path) -> Tuple[float, float, float, float]:
    """计算路径边界框 (min_x, min_y, max_x, max_y)"""
    if not path.points:
        return (0, 0, 0, 0)

    xs = [p.x for p in path.points]
    ys = [p.y for p in path.points]
    return (min(xs), min(ys), max(xs), max(ys))

def is_clockwise(path: Path) -> bool:
    """判断路径是否顺时针方向"""
    area = 0
    n = len(path.points)
    for i in range(n):
        j = (i + 1) % n
        area += path.points[i].x * path.points[j].y
        area -= path.points[j].x * path.points[i].y
    return area < 0

def reverse_path(path: Path) -> Path:
    """反转路径方向"""
    return Path(
        points=list(reversed(path.points)),
        path_type=path.path_type,
        color=path.color
    )

def point_in_polygon(point: Point, polygon: Path) -> bool:
    """射线法判断点是否在多边形内"""
    x, y = point.x, point.y
    n = len(polygon.points)
    inside = False

    j = n - 1
    for i in range(n):
        xi, yi = polygon.points[i].x, polygon.points[i].y
        xj, yj = polygon.points[j].x, polygon.points[j].y

        if ((yi > y) != (yj > y)) and (x < (xj - xi) * (y - yi) / (yj - yi) + xi):
            inside = not inside
        j = i

    return inside

def rotate_point(point: Point, center: Point, angle_deg: float) -> Point:
    """绕中心点旋转"""
    rad = math.radians(angle_deg)
    cos_a = math.cos(rad)
    sin_a = math.sin(rad)

    dx = point.x - center.x
    dy = point.y - center.y

    new_x = center.x + dx * cos_a - dy * sin_a
    new_y = center.y + dx * sin_a + dy * cos_a

    return Point(new_x, new_y)

def rotate_path(path: Path, center: Point, angle_deg: float) -> Path:
    """旋转整个路径"""
    rotated_points = [rotate_point(p, center, angle_deg) for p in path.points]
    return Path(
        points=rotated_points,
        path_type=path.path_type,
        color=path.color
    )

def scale_path(path: Path, center: Point, scale: float) -> Path:
    """缩放路径"""
    scaled_points = [
        Point(
            center.x + (p.x - center.x) * scale,
            center.y + (p.y - center.y) * scale
        )
        for p in path.points
    ]
    return Path(
        points=scaled_points,
        path_type=path.path_type,
        color=path.color
    )

def translate_path(path: Path, dx: float, dy: float) -> Path:
    """平移路径"""
    translated_points = [Point(p.x + dx, p.y + dy) for p in path.points]
    return Path(
        points=translated_points,
        path_type=path.path_type,
        color=path.color
    )

def simplify_path_rdp(path: Path, epsilon: float) -> Path:
    """
    Douglas-Peucker算法简化路径

    Args:
        path: 输入路径
        epsilon: 简化阈值（距离）

    Returns:
        简化后的路径
    """
    if len(path.points) <= 2:
        return path

    # 找到距离首尾连线最远的点
    start = path.points[0]
    end = path.points[-1]

    max_dist = 0
    max_index = 0

    for i in range(1, len(path.points) - 1):
        dist = point_to_line_distance(path.points[i], start, end)
        if dist > max_dist:
            max_dist = dist
            max_index = i

    # 如果最大距离大于阈值，递归简化
    if max_dist > epsilon:
        left_path = Path(points=path.points[:max_index + 1])
        right_path = Path(points=path.points[max_index:])

        left_simplified = simplify_path_rdp(left_path, epsilon)
        right_simplified = simplify_path_rdp(right_path, epsilon)

        return Path(
            points=left_simplified.points[:-1] + right_simplified.points,
            path_type=path.path_type,
            color=path.color
        )
    else:
        return Path(
            points=[start, end],
            path_type=path.path_type,
            color=path.color
        )

def point_to_line_distance(point: Point, line_start: Point, line_end: Point) -> float:
    """计算点到线段的距离"""
    dx = line_end.x - line_start.x
    dy = line_end.y - line_start.y

    if dx == 0 and dy == 0:
        return distance(point, line_start)

    t = max(0, min(1, (
        (point.x - line_start.x) * dx +
        (point.y - line_start.y) * dy
    ) / (dx * dx + dy * dy)))

    proj = Point(
        line_start.x + t * dx,
        line_start.y + t * dy
    )

    return distance(point, proj)
```
